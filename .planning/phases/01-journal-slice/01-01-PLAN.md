      _HABER_COL_CANDIDATES: tuple[str, ...] = ("eurohaber", "eurhaber", "haber", "pesehaber")

    _pick_column(field_set: set[str], candidates: tuple[str, ...], *, kind: str) -> str:
      Iterate candidates; return first match in field_set. On miss, raise
      ContaPlusReadError(row_index=-1, column=kind, message="DBF has no {kind} column; tried {candidates}; available: {sorted(field_set)}")

    _assert_journal_shaped(table: DBF) -> None:
      Implements D-02 — checks the DBF field signature BEFORE journal parsing so a
      valid-but-non-journal DBF is rejected with a scope-aware error, not a generic column failure.
      Build field_set = {n.lower() for n in table.field_names}
      Build field_type_map = {f.name.lower(): f.type for f in table.fields}
      Check "fecha" in field_set AND field_type_map.get("fecha") == "D"
      Check field_set intersects {"eurodebe","eurdebe","debe","pesedebe"}
      Check field_set intersects {"eurohaber","eurhaber","haber","pesehaber"}
      On any failure raise ContaPlusReadError(row_index=-1, column=None, message=
        "Input is a valid DBF but not a DIARIO.DBF journal table (multi-table support is planned in a future phase). Available fields: {sorted(field_set)}")

    _read_dbf_path(path: Path, *, source_name: str | None = None) -> ContaPlusJournal:
      Wrap in try/except. Open with:
        DBF(str(path), lowernames=True, encoding="cp850", ignore_missing_memofile=True)
      Call _assert_journal_shaped(table) before _pick_column calls.
      Resolve debe_col and haber_col via _pick_column.
      Iterate records (lazy, do not pass load=True):
        Per record order: D-C4 null fecha → D-C2 both-non-zero → D-C3 both-zero → D-E3 subcuenta
        D-C2 guard is "debe > 0 and haber > 0" NOT "debe != 0 and haber != 0" (Pitfall 2)
        D-E3: subcuenta = record.get("subcta") stripped with .rstrip(); must be non-empty, isdigit(), len>=3
        D-B2: cuenta = subcuenta[:4]
        concepto: rstrip(); empty string becomes None
        debe/haber: float(record.get(col) or 0) — handles Decimal and None from dbfread
      Collect list[JournalRow]. After loop, logger.info if skipped_memo > 0.
      Return ContaPlusJournal(rows=rows, skipped_memo=skipped_memo, source_name=source_name)
      Except clause: re-raise ContaPlusReadError unchanged; wrap UnicodeDecodeError,
      struct.error, ValueError, OSError into ContaPlusReadError(row_index=-1, ..., original=exc).
